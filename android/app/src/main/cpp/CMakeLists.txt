cmake_minimum_required(VERSION 3.22.1)
project(echocraft)
include(FetchContent)
set(LAME_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.deps/lame-3.100")
if(NOT EXISTS "${LAME_ROOT}/include/lame.h")
    FetchContent_Declare(lame_source
        URL https://downloads.sourceforge.net/project/lame/lame/3.100/lame-3.100.tar.gz
            https://download.videolan.org/pub/contrib/lame/lame-3.100.tar.gz
        URL_HASH SHA256=ddfe36cab873794038ae2c1210557ad34857a4b6bdc515785d1da9e175b1da1e)
    FetchContent_GetProperties(lame_source)
    if(NOT lame_source_POPULATED)
        FetchContent_Populate(lame_source)
    endif()
    set(LAME_ROOT "${lame_source_SOURCE_DIR}")
endif()
set(LAME_FILES VbrTag bitstream encoder fft gain_analysis id3tag lame newmdct presets
    psymodel quantize quantize_pvt reservoir set_get tables takehiro util vbrquantize version mpglib_interface)
set(LAME_SOURCES "")
foreach(name IN LISTS LAME_FILES)
    list(APPEND LAME_SOURCES "${LAME_ROOT}/libmp3lame/${name}.c")
endforeach()
add_library(mp3lame SHARED ${LAME_SOURCES})
target_include_directories(mp3lame PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/lame-config "${LAME_ROOT}/include" "${LAME_ROOT}/libmp3lame")
target_compile_definitions(mp3lame PRIVATE HAVE_CONFIG_H=1)
target_link_libraries(mp3lame m)
target_link_options(mp3lame PRIVATE "-Wl,-z,max-page-size=16384")
add_library(echocraft SHARED audio_jni.cpp mp3_jni.cpp)
target_include_directories(echocraft PRIVATE "${LAME_ROOT}/include")
target_link_libraries(echocraft mp3lame)
target_compile_features(echocraft PRIVATE cxx_std_17)
target_include_directories(echocraft PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../../../../firmware/mic-test/src)
